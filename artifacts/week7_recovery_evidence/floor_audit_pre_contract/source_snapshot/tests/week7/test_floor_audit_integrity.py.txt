"""Independent integrity checks for the read-only paired-floor auditor.

The native archive is the one cached fixed R=2 validation archive shared with
the mathematical review. Every archive fault below changes a private copy.
Literal failure-stream and coordinate fixtures exercise only those pure
validators; they are never represented as simulated or full replay evidence.
"""

def encoded(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':'), allow_nan=False)+'\n').encode()


def digest(value):
    return hashlib.sha256(value).hexdigest()


def snapshot(output):
    result = {}
    for path in sorted(output.rglob('*')):
        key = str(path.relative_to(output))
        if path.is_symlink():
            result[key] = ('symlink', str(path.readlink()))
        elif path.is_file():
            result[key] = ('file', digest(path.read_bytes()))
        elif path.is_dir():
            result[key] = ('directory',)
    return result


def read_json(path):
    return json.loads(path.read_bytes())


def write_json(path, value):
    path.write_bytes(encoded(value))


def forbidden(*args, **kwargs):
    raise AssertionError('read-only audit must not invoke raw execution or mutate artifacts')


@pytest.fixture(scope='module')
def native(tmp_path_factory):
    return native_archive(tmp_path_factory)


@pytest.fixture
def copied(native, tmp_path):
    output = tmp_path/'copied_archive'
    shutil.copytree(native['output'], output)
    return output, native['design']


@pytest.fixture(scope='module')
def complete_native_report(native):
    """One additional deterministic full-fit replay for lifecycle seal tests."""
    before = snapshot(native['output'])
    report = floor_audit.audit_run(native['output'], native['design'], replay=True)
    assert report['full_completion_audit'] is True
    assert snapshot(native['output']) == before
    return report


def reject_unchanged(output, design, match=None):
    before = snapshot(output)
    with pytest.raises(ValueError, match=match):
        floor_audit.audit_run(output, design, replay=False)
    assert snapshot(output) == before


def first_cell(design):
    return next(c for c in design.cells if c.cv == 1.5 and c.floor_fraction == 1e-8)


def rebind_raw_inputs(output):
    """Rehash a fault copy so its rejection cannot rely on a stale EPS binding."""
    manifest_path = output/'epsilon_table_manifest.json'
    manifest = read_json(manifest_path)
    metadata = read_json(output/'run_metadata.json')
    binding = manifest['raw_inputs']
    binding['run_identity'] = metadata['provenance']['identity_sha256']
    binding['source_policy'] = metadata['source_policy']
    binding['resolved_design_sha256'] = digest((output/'resolved_design.json').read_bytes())
    for cid in binding['cells']:
        path = output/'cells'/cid/'cell_manifest.json'
        state = read_json(path)
        binding['cells'][cid] = {'manifest_sha256': digest(path.read_bytes()),
                                 'partitions': deepcopy(state['partitions'])}
    manifest['run_identity'] = binding['run_identity']
    manifest['raw_input_sha256'] = digest(encoded(binding).rstrip(b'\n'))
    write_json(manifest_path, manifest)


def mutate_raw(output, design, change, *, text_change=None):
    cell = first_cell(design)
    directory = output/'cells'/cell.identity
    state_path = directory/'cell_manifest.json'
    state = read_json(state_path)
    part = state['partitions'][0]
    path = directory/part['filename']
    payload = gzip.decompress(path.read_bytes())
    rows = [json.loads(line) for line in payload.splitlines()]
    change(rows)
    payload = b''.join(encoded(row) for row in rows)
    if text_change is not None:
        payload = text_change(payload)
    path.write_bytes(gzip.compress(payload, mtime=0))
    part['sha256'] = digest(path.read_bytes())
    write_json(state_path, state)
    rebind_raw_inputs(output)
    return rows


def mutate_table(output, name, change):
    path = output/name
    table = pd.read_csv(path, float_precision='round_trip')
    change(table)
    table.to_csv(path, index=False, float_format='%.17g', lineterminator='\n')
    if name == 'epsilon_comparisons.csv':
        manifest_path = output/'epsilon_table_manifest.json'
        manifest = read_json(manifest_path)
        manifest['table']['sha256'] = digest(path.read_bytes())
    else:
        manifest_path = output/'table_manifest.json'
        manifest = read_json(manifest_path)
        manifest[name]['sha256'] = digest(path.read_bytes())
    write_json(manifest_path, manifest)


def mutate_metadata(output, change):
    path = output/'run_metadata.json'
    metadata = read_json(path)
    change(metadata)
    write_json(path, metadata)

@pytest.mark.parametrize('status', ('audited_pending_visual_review', 'completed'))
def test_closed_archive_is_verified_without_raw_preflight_or_any_write(
        copied, complete_native_report, monkeypatch, status):
    output, design = copied
    write_json(output/'audit.json', complete_native_report)
    metadata = read_json(output/'run_metadata.json')
    metadata.update(status=status, audit_sha256=digest((output/'audit.json').read_bytes()))
    if status == 'completed':
        # Literal, private lifecycle fixtures test seal validation only. They
        # are not scientific figures or a claim of actual visual inspection.
        figures = output/'figures'
        figures.mkdir()
        panel = figures/'literal_test_panel.png'
        panel.write_bytes(b'literal non-scientific figure-byte fixture\n')
        figure = figures/'figure_manifest.json'
        write_json(figure, {'figures': {panel.name: digest(panel.read_bytes())}})
        review = output/'visual_review.json'
        write_json(review, {'all_panels_legible': True, 'reviewed_pngs': [panel.name],
                            'reviewer': 'literal lifecycle unit fixture; no scientific visual review',
                            'figure_manifest_sha256': digest(figure.read_bytes())})
        metadata['visual_review_sha256'] = digest(review.read_bytes())
        metadata['artifact_sha256'] = {
            str(path.relative_to(output)): digest(path.read_bytes())
            for path in output.rglob('*') if path.is_file()
            and path.name not in {'run_metadata.json', '.execution.lock'}}
    write_json(output/'run_metadata.json', metadata)
    before = snapshot(output)
    monkeypatch.setattr(execution, 'preflight', forbidden)
    monkeypatch.setattr(execution, '_output_path', forbidden)
    monkeypatch.setattr(execution, 'run_raw', forbidden)
    report = floor_audit.audit_run(output, design, replay=False)
    assert report['status'] == 'passed' and report['full_completion_audit'] is False
    assert snapshot(output) == before

@pytest.mark.parametrize('fault', ('missing_cell', 'extra_cell', 'extra_root_file',
                                  'extra_partition', 'partition_symlink', 'cell_symlink'))
def test_strict_complete_layout_and_symlinks(copied, tmp_path, fault):
    output, design = copied
    cell = first_cell(design)
    directory = output/'cells'/cell.identity
    part = next(directory.glob('*.jsonl.gz'))
    if fault == 'missing_cell':
        shutil.rmtree(directory)
    elif fault == 'extra_cell':
        (output/'cells'/'unexpected').mkdir()
    elif fault == 'extra_root_file':
        (output/'unregistered.txt').write_text('unexpected evidence\n')
    elif fault == 'extra_partition':
        (directory/'part-000100-000101.jsonl.gz').write_bytes(part.read_bytes())
    elif fault == 'partition_symlink':
        target = tmp_path/'outside_partition.gz'
        target.write_bytes(part.read_bytes())
        part.unlink()
        part.symlink_to(target)
    elif fault == 'cell_symlink':
        target = tmp_path/'outside_cell'
        directory.rename(target)
        directory.symlink_to(target, target_is_directory=True)
    reject_unchanged(output, design)

def test_consistent_forged_observed_offset_is_detected_from_actual_arrays(copied):
    output, design = copied
    def change(rows):
        row = rows[0]
        point = json.loads(row['point_diagnostics_json'])
        interval = json.loads(row['interval_diagnostics_json'])
        # Preserve point/interval agreement and normalized eta algebra. The
        # independent observed-data calculation must still reject the forgery.
        delta = .01
        point['state_center'] += delta
        interval['state_offset'] += delta
        interval['working_parameters'][1] -= delta/interval['state_scale']
        steps = interval['hessian_diagnostics']['steps']
        steps[1] = .001*max(1., abs(interval['working_parameters'][1]))
        row['point_diagnostics_json'] = json.dumps(point, sort_keys=True, separators=(',', ':'))
        row['interval_diagnostics_json'] = json.dumps(interval, sort_keys=True, separators=(',', ':'))
    mutate_raw(output, design, change)
    reject_unchanged(output, design, 'observed|coordinate|replay')

def test_root_drift_is_observed_without_repair_or_incident_writes(copied, monkeypatch):
    output, design = copied
    path = output/'run_metadata.json'
    original = OuSampleSource.sample
    changed = None
    def drift(source, cell, rep):
        nonlocal changed
        result = original(source, cell, rep)
        if changed is None:
            metadata = read_json(path)
            metadata['status'] = 'unexpected_external_state'
            changed = encoded(metadata)
            path.write_bytes(changed)
        return result
    monkeypatch.setattr(OuSampleSource, 'sample', drift)
    with pytest.raises(ValueError, match='changed|drift'):
        floor_audit.audit_run(output, design, replay=False)
    assert path.read_bytes() == changed
    assert not list(output.parent.glob(output.name+'-manifest-incident-*'))


def test_non_numerical_existing_artifact_drift_is_in_complete_readback_scope(
        copied, complete_native_report, monkeypatch):
    output, design = copied
    # audit.json is a permitted later lifecycle artifact. Its bytes still
    # belong to this invocation's complete before/after read-only snapshot.
    path = output/'audit.json'
    write_json(path, complete_native_report)
    original = OuSampleSource.sample
    changed = encoded({'prior_audit': 'externally replaced'})
    touched = False
    def drift(source, cell, rep):
        nonlocal touched
        result = original(source, cell, rep)
        if not touched:
            path.write_bytes(changed)
            touched = True
        return result
    monkeypatch.setattr(OuSampleSource, 'sample', drift)
    with pytest.raises(ValueError, match='changed|drift'):
        floor_audit.audit_run(output, design, replay=False)
    assert path.read_bytes() == changed


def rehash_provenance(provenance_record):
    identity = {key: provenance_record[key] for key in (
        'design', 'design_sha256', 'preregistration_commit', 'source_files', 'environment')}
    provenance_record['identity_sha256'] = digest(encoded(identity).rstrip(b'\n'))

def write_failure_rows(path, rows, *, header=None):
    header = sorted(rows[0]) if header is None else header
    text = io.StringIO(newline='')
    writer = csv.writer(text, lineterminator='\n')
    writer.writerow(header)
    for row in rows:
        writer.writerow([row.get(name, '') for name in header])
    path.write_bytes(gzip.compress(text.getvalue().encode(), mtime=0))

    rows[0].append(rows[0][0])
